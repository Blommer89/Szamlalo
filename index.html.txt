<script>
    const gyakorlatok = ['fekvo', 'guggolas', 'labemeles'];

    function getTodayString() {
        const d = new Date();
        return d.toISOString().split('T')[0];
    }

    function frissitOldal() {
        const today = getTodayString();

        gyakorlatok.forEach(id => {
            let mentettCel = localStorage.getItem('cel_' + id);
            if (mentettCel) {
                document.getElementById('target_' + id).value = mentettCel;
            }

            let rawHistory = localStorage.getItem('history_' + id);
            let history = rawHistory ? JSON.parse(rawHistory) : {};

            let maiErtek = history[today] || 0;
            document.getElementById('val_' + id).textContent = maiErtek;

            let histHtml = '';
            let sortedDates = Object.keys(history).sort().reverse();
            
            if (sortedDates.length === 0) {
                histHtml = '<div style="color:#64748b; text-align:center;">Még nincs rögzített nap.</div>';
            } else {
                sortedDates.forEach(d => {
                    histHtml += `<div class="history-row"><span>${d === today ? 'Ma (' + d + ')' : d}</span><span><b>${history[d]} db</b></span></div>`;
                });
            }
            document.getElementById('hist_box_' + id).innerHTML = histHtml;
        });
    }

    function novel(id) {
        const today = getTodayString();
        let rawHistory = localStorage.getItem('history_' + id);
        let history = rawHistory ? JSON.parse(rawHistory) : {};

        let jelenlegi = history[today] || 0;
        history[today] = jelenlegi + 1;

        localStorage.setItem('history_' + id, JSON.stringify(history));
        frissitOldal();
    }

    function toggleNaptar(id) {
        const box = document.getElementById('hist_box_' + id);
        box.classList.toggle('active');
    }

    gyakorlatok.forEach(id => {
        document.getElementById('target_' + id).addEventListener('change', function() {
            localStorage.setItem('cel_' + id, this.value);
        });
    });

    frissitOldal();
</script>
